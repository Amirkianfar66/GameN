npm ci
npm run build --workspace apps/game
npm run build --workspace packages/engine
node apps/game/dev/phone/food-comparison-plan.mjs --check
node apps/game/dev/phone/food-native-packets.mjs
git worktree add --detach /private/tmp/gamen-food-art-base-1f795366 1f795366a27b17f46ac195debe491954a27d8ae0
GAMEN_PHONE_ART_BASE=/private/tmp/gamen-food-art-base-1f795366 node_modules/.bin/vite build --config apps/game/dev/phone/food-vite.config.mjs
node apps/game/dev/phone/food-app-smoke.mjs
node apps/game/dev/phone/food-compare.mjs
GAMEN_PHONE_ART_BASE=/private/tmp/gamen-food-art-base-1f795366 node_modules/.bin/vite build --config apps/game/dev/phone/vite.config.mjs --outDir /private/tmp/gamen-food-ordinary-review
node apps/game/dev/phone/food-ordinary-compare.mjs

bypass current-tree production admission. Required CI, combined release digest,
Designer approval and staging deployment remain Integration's separate gates.
